SELECT COUNT(*) AS Total_Vacant_Rooms FROM ROOM  WHERE Status = 'Vacant';

SELECT H.Hno, H.Hname, COUNT(S.Sid) AS Total_Students FROM HOSTEL H
JOIN STUDENT S ON H.Hno = S.Hno
JOIN ROOM R ON S.Hno = R.Hno AND S.Rno = R.Rno
WHERE R.Rtype = 'Double' GROUP BY H.Hno, H.Hname;

SELECT H.Warden_Nm, H.Haddress FROM HOSTEL H JOIN STUDENT S ON H.Hno = S.Hno WHERE S.Faculty = 'Computer Science';


SELECT DISTINCT H.*
FROM HOSTEL H JOIN ROOM R ON H.Hno = R.Hno
WHERE R.Status = 'Vacant' AND R.Rtype IN ('Single', 'Four');

SELECT COUNT(*) AS Total_Medical_Students
FROM STUDENT
WHERE Faculty = 'Medical';


SELECT S.*
FROM STUDENT S
JOIN ROOM R ON S.Hno = R.Hno AND S.Rno = R.Rno
JOIN HOSTEL H ON R.Hno = H.Hno
WHERE H.Hname = 'Chanakya' AND R.Rtype = 'Double';
  
 SELECT H.Hname, R.Rtype, COUNT(S.Sid) AS Total_Students
FROM HOSTEL H JOIN ROOM R ON H.Hno = R.Hno
JOIN STUDENT S ON R.Hno = S.Hno AND R.Rno = S.Rno
GROUP BY H.Hno, H.Hname, R.Rtype;

SELECT S.* FROM STUDENT S JOIN FEES F ON S.Sid = F.Sid
